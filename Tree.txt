
import streamlit as st
import pandas as pd
import joblib

st.set_page_config(
    page_title="Decision Tree Predictor",
    page_icon="🌳",
    layout="wide"
)

st.title("🌳 Decision Tree Classifier")
st.write("Upload a CSV file to generate predictions.")

@st.cache_resource
def load_model():
    return joblib.load("decision_tree_model.joblib")

try:
    saved = load_model()
    model = saved["model"]
    feature_columns = saved["feature_columns"]
except Exception as e:
    st.error(f"Could not load model: {e}")
    st.stop()

st.success("Decision Tree model loaded successfully!")

uploaded_file = st.file_uploader(
    "Upload your CSV file",
    type=["csv"]
)

if uploaded_file is not None:
    try:
        data = pd.read_csv(uploaded_file)

        st.subheader("Uploaded Data")
        st.dataframe(data.head())

        missing = [
            col for col in feature_columns
            if col not in data.columns
        ]

        if missing:
            st.error(
                "Your CSV is missing required columns: "
                + ", ".join(missing)
            )
        else:
            # Use exactly the feature columns from training
            X_new = data[feature_columns].copy()

            predictions = model.predict(X_new)

            results = data.copy()
            results["Prediction"] = predictions

            st.subheader("Prediction Results")
            st.dataframe(results)

            csv = results.to_csv(index=False).encode("utf-8")

            st.download_button(
                "Download Predictions",
                data=csv,
                file_name="decision_tree_predictions.csv",
                mime="text/csv"
            )

    except Exception as e:
        st.error(f"Error processing file: {e}")
